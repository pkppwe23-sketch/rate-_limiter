cmake_minimum_required(VERSION 3.16)
project(rate_limiter)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(rate_limiter src/RateLimiter.cpp)
target_include_directories(rate_limiter PUBLIC src)
add_executable(rate_limiter_demo src/main.cpp)
target_link_libraries(rate_limiter_demo PRIVATE rate_limiter)
add_executable(rate_limiter_tests tests/test_rate_limiter.cpp)
target_link_libraries(rate_limiter_tests PRIVATE rate_limiter)
enable_testing()
add_test(NAME rate_limiter_tests COMMAND rate_limiter_tests)
